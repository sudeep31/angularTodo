import { ComponentFixture, TestBed } from '@angular/core/testing';
import { describe, it, expect, beforeEach, vi } from 'vitest';
import { TodoListComponent } from './todo-list.component';
import { TodoItemComponent } from '../../components/todo-item/todo-item.component';
import { Todo } from '../../interfaces/todo.interface';

describe('TodoListComponent', () => {
  let component: TodoListComponent;
  let fixture: ComponentFixture<TodoListComponent>;

  // Helper function to create mock input events
  const createInputEvent = (value: string): Event => {
    return Object.defineProperty({}, 'target', {
      value: { value },
      enumerable: true
    }) as Event;
  };

  // Helper function to create mock keyboard events
  const createKeyboardEvent = (key: string): KeyboardEvent => {
    return Object.defineProperty({}, 'key', {
      value: key,
      enumerable: true
    }) as KeyboardEvent;
  };

  beforeEach(async () => {
    await TestBed.configureTestingModule({
      imports: [TodoListComponent, TodoItemComponent]
    }).compileComponents();

    fixture = TestBed.createComponent(TodoListComponent);
    component = fixture.componentInstance;
    fixture.detectChanges();
  });

  describe('Component Creation & Initialization', () => {
    it('should create', () => {
      expect(component).toBeTruthy();
    });

    it('should initialize with sample data', () => {
      expect(component.testTodos().length).toBeGreaterThan(0);
    });

    it('should have proper initial state', () => {
      expect(component.testFilter()).toBe('all');
      expect(component.testLoading()).toBe(false);
      expect(component.testNewTodoForm().title).toBe('');
      expect(component.testNewTodoForm().description).toBe('');
      expect(component.testNewTodoForm().priority).toBe('medium');
      expect(component.testNewTodoForm().dueDate).toBe('');
    });

    it('should have initial form errors as empty', () => {
      expect(component.testFormErrors().title).toBe('');
      expect(component.testFormErrors().description).toBe('');
    });

    it('should have form initially invalid', () => {
      expect(component.testIsFormValid()).toBe(false);
    });
  });

  describe('Form Validation Scenarios', () => {
    describe('Title Field Validation', () => {
      it('should show required error when title is empty', () => {
        const event = createInputEvent('');
        component.onTitleInput(event);

        expect(component.testFormErrors().title).toBe('Title is required');
        expect(component.testIsFormValid()).toBe(false);
      });

      it('should show pattern error when title contains special characters', () => {
        const event = createInputEvent('Invalid @#$%');
        component.onTitleInput(event);

        expect(component.testFormErrors().title).toBe('Title should only contain letters, numbers, and spaces');
        expect(component.testIsFormValid()).toBe(false);
      });

      it('should clear error when valid title is entered', () => {
        // First set invalid title
        const invalidEvent = createInputEvent('Invalid @#$%');
        component.onTitleInput(invalidEvent);
        expect(component.testFormErrors().title).not.toBe('');

        // Then set valid title
        const validEvent = createInputEvent('Valid Title');
        component.onTitleInput(validEvent);

        expect(component.testFormErrors().title).toBe('');
        expect(component.testNewTodoForm().title).toBe('Valid Title');
      });

      it('should handle whitespace-only title as empty', () => {
        const event = createInputEvent('   ');
        component.onTitleInput(event);

        expect(component.testFormErrors().title).toBe('Title is required');
      });
    });

    describe('Description Field Validation', () => {
      it('should show required error when description is empty', () => {
        const event = createInputEvent('');
        component.onDescriptionInput(event);

        expect(component.testFormErrors().description).toBe('Description is required');
        expect(component.testIsFormValid()).toBe(false);
      });

      it('should show pattern error when description contains special characters', () => {
        const event = createInputEvent('Invalid description @#$%');
        component.onDescriptionInput(event);

        expect(component.testFormErrors().description).toBe('Description should only contain letters, numbers, and spaces');
        expect(component.testIsFormValid()).toBe(false);
      });

      it('should clear error when valid description is entered', () => {
        // First set invalid description
        const invalidEvent = createInputEvent('Invalid @#$%');
        component.onDescriptionInput(invalidEvent);
        expect(component.testFormErrors().description).not.toBe('');

        // Then set valid description
        const validEvent = createInputEvent('Valid description text');
        component.onDescriptionInput(validEvent);

        expect(component.testFormErrors().description).toBe('');
        expect(component.testNewTodoForm().description).toBe('Valid description text');
      });
    });

    describe('Form Validity Scenarios', () => {
      it('should be valid when all required fields are valid', () => {
        const titleEvent = createInputEvent('Test Todo');
        const descriptionEvent = createInputEvent('Test description');

        component.onTitleInput(titleEvent);
        component.onDescriptionInput(descriptionEvent);

        expect(component.testIsFormValid()).toBe(true);
      });

      it('should be invalid if only title is valid', () => {
        const titleEvent = createInputEvent('Test Todo');
        component.onTitleInput(titleEvent);

        expect(component.testIsFormValid()).toBe(false);
      });

      it('should be invalid if only description is valid', () => {
        const descriptionEvent = createInputEvent('Test description');
        component.onDescriptionInput(descriptionEvent);

        expect(component.testIsFormValid()).toBe(false);
      });
    });
  });

  describe('Form Input Handling Scenarios', () => {
    describe('Priority Selection', () => {
      it('should update priority when low is selected', () => {
        const event = createInputEvent('low');
        component.onPriorityChange(event);

        expect(component.testNewTodoForm().priority).toBe('low');
      });

      it('should update priority when high is selected', () => {
        const event = createInputEvent('high');
        component.onPriorityChange(event);

        expect(component.testNewTodoForm().priority).toBe('high');
      });

      it('should maintain medium as default priority', () => {
        expect(component.testNewTodoForm().priority).toBe('medium');
      });

      it('should update due date when valid date is entered', () => {
        const event = createInputEvent('2024-12-31');
        component.onDueDateChange(event);

        expect(component.testNewTodoForm().dueDate).toBe('2024-12-31');
      });

      it('should handle empty due date', () => {
        const event = createInputEvent('');
        component.onDueDateChange(event);

        expect(component.testNewTodoForm().dueDate).toBe('');
      });
    });

    describe('Keyboard Navigation', () => {
      it('should add todo when Enter key is pressed with valid form', () => {
        // Set up valid form
        const titleEvent = createInputEvent('Test Todo');
        const descriptionEvent = createInputEvent('Test description');
        component.onTitleInput(titleEvent);
        component.onDescriptionInput(descriptionEvent);

        const initialLength = component.testTodos().length;
        const enterEvent = createKeyboardEvent('Enter');

        component.onFormKeydown(enterEvent);

        expect(component.testTodos()).toHaveLength(initialLength + 1);
        expect(component.testTodos()[component.testTodos().length - 1].title).toBe('Test Todo');
        expect(component.testTodos()[component.testTodos().length - 1].description).toBe('Test description');
      });

      it('should not add todo when Enter key is pressed with invalid form', () => {
        const initialLength = component.testTodos().length;
        const enterEvent = createKeyboardEvent('Enter');

        component.onFormKeydown(enterEvent);

        expect(component.testTodos()).toHaveLength(initialLength);
      });

      it('should ignore other keys', () => {
        const initialLength = component.testTodos().length;
        const escapeEvent = createKeyboardEvent('Escape');

        component.onFormKeydown(escapeEvent);

        expect(component.testTodos()).toHaveLength(initialLength);
      });
    });
  });

      component.testTodos.set([]);

      const descriptionEvent = createInputEvent('Test description');
      const priorityEvent = createInputEvent('high');
      const dueDateEvent = createInputEvent('2024-12-31');

      component.onTitleInput(titleEvent);
      component.onDescriptionInput(descriptionEvent);
      component.onPriorityChange(priorityEvent);
      component.onDueDateChange(dueDateEvent);

      component.onAddTodo();

      expect(component.testTodos()).toHaveLength(1);
      const addedTodo = component.testTodos()[0];
      expect(addedTodo.title).toBe('Test Todo');
      expect(addedTodo.description).toBe('Test description');
      expect(addedTodo.priority).toBe('high');
      expect(addedTodo.completed).toBe(false);
      expect(addedTodo.dueDate).toEqual(new Date('2024-12-31'));
      expect(addedTodo.id).toBeTruthy();
      expect(addedTodo.createdAt).toBeInstanceOf(Date);
      expect(addedTodo.updatedAt).toBeInstanceOf(Date);
    });

    it('should add todo with required fields only when optional fields are empty', () => {
      const titleEvent = createInputEvent('Test Todo');
      const descriptionEvent = createInputEvent('Test description');

      component.onTitleInput(titleEvent);
      component.onDescriptionInput(descriptionEvent);

      component.onAddTodo();

      expect(component.testTodos()).toHaveLength(1);
      const addedTodo = component.testTodos()[0];
      expect(addedTodo.title).toBe('Test Todo');
      expect(addedTodo.description).toBe('Test description');
      expect(addedTodo.priority).toBe('medium'); // default value
      expect(addedTodo.dueDate).toBeUndefined();
    });

    it('should trim whitespace from title and description', () => {
      const titleEvent = createInputEvent('  Test Todo  ');
      const descriptionEvent = createInputEvent('  Test description  ');

      component.onTitleInput(titleEvent);
      component.onDescriptionInput(descriptionEvent);

      component.onAddTodo();

      const addedTodo = component.testTodos()[0];
      expect(addedTodo.title).toBe('Test Todo');
      expect(addedTodo.description).toBe('Test description');
    });

    it('should not add todo when form is invalid due to empty title', () => {
      const descriptionEvent = createInputEvent('Test description');
      component.onDescriptionInput(descriptionEvent);

      // Mock console.log to verify error message
      const consoleSpy = vi.spyOn(console, 'log').mockImplementation(() => { });

      component.onAddTodo();

      expect(component.testTodos()).toHaveLength(0);
      expect(consoleSpy).toHaveBeenCalledWith('Form is invalid, cannot submit');

      consoleSpy.mockRestore();
    });

    it('should not add todo when form is invalid due to empty description', () => {
      const titleEvent = createInputEvent('Test Todo');
      component.onTitleInput(titleEvent);

      const consoleSpy = vi.spyOn(console, 'log').mockImplementation(() => { });

      component.onAddTodo();

      expect(component.testTodos()).toHaveLength(0);
      expect(consoleSpy).toHaveBeenCalledWith('Form is invalid, cannot submit');

      consoleSpy.mockRestore();
    });

    it('should reset form after successful todo addition', () => {
      const titleEvent = createInputEvent('Test Todo');
      const descriptionEvent = createInputEvent('Test description');
      const priorityEvent = createInputEvent('high');
      const dueDateEvent = createInputEvent('2024-12-31');

      component.onTitleInput(titleEvent);
      component.onDescriptionInput(descriptionEvent);
      component.onPriorityChange(priorityEvent);
      component.onDueDateChange(dueDateEvent);

      component.onAddTodo();

      // Check form is reset
      expect(component.testNewTodoForm().title).toBe('');
      expect(component.testNewTodoForm().description).toBe('');
      expect(component.testNewTodoForm().priority).toBe('medium');
      expect(component.testNewTodoForm().dueDate).toBe('');
      expect(component.testFormErrors().title).toBe('');
      expect(component.testFormErrors().description).toBe('');
    });
  });

  describe('Todo Management Scenarios', () => {
    let testTodos: Todo[];

    beforeEach(() => {
      testTodos = [
        {
          id: '1',
          title: 'Test Todo 1',
          description: 'Description 1',
          completed: false,
          priority: 'high',
          createdAt: new Date(),
          updatedAt: new Date(),
          tags: []
        },
        {
          id: '2',
          title: 'Test Todo 2',
          description: 'Description 2',
          completed: true,
          priority: 'low',
          createdAt: new Date(),
          updatedAt: new Date(),
          tags: []
        }
      ];
      component.testTodos.set(testTodos);
    });

    describe('Todo Toggling', () => {
      it('should toggle incomplete todo to complete', () => {
        const originalUpdatedAt = component.testTodos()[0].updatedAt;

        component.onTodoToggled('1');

        expect(component.testTodos()[0].completed).toBe(true);
        expect(component.testTodos()[0].updatedAt).not.toEqual(originalUpdatedAt);
        expect(component.testTodos()[1].completed).toBe(true); // Unchanged
      });

      it('should toggle complete todo to incomplete', () => {
        const originalUpdatedAt = component.testTodos()[1].updatedAt;

        component.onTodoToggled('2');

        expect(component.testTodos()[1].completed).toBe(false);
        expect(component.testTodos()[1].updatedAt).not.toEqual(originalUpdatedAt);
        expect(component.testTodos()[0].completed).toBe(false); // Unchanged
      });

      it('should not affect other todos when toggling one', () => {
        const originalTodo2 = { ...component.testTodos()[1] };

        component.onTodoToggled('1');

        expect(component.testTodos()[1]).toEqual(originalTodo2);
      });

      it('should handle toggling non-existent todo gracefully', () => {
        const originalTodos = [...component.testTodos()];

        component.onTodoToggled('non-existent-id');

        expect(component.testTodos()).toEqual(originalTodos);
      });
    });

    describe('Todo Deletion', () => {
      it('should delete todo by id', () => {
        component.onTodoDeleted('1');

        expect(component.testTodos()).toHaveLength(1);
        expect(component.testTodos()[0].id).toBe('2');
        expect(component.testTodos().find(todo => todo.id === '1')).toBeUndefined();
      });

      it('should handle deletion of non-existent todo gracefully', () => {
        const originalLength = component.testTodos().length;

        component.onTodoDeleted('non-existent-id');

        expect(component.testTodos()).toHaveLength(originalLength);
      });

      it('should delete all todos when called for each one', () => {
        component.onTodoDeleted('1');
        component.onTodoDeleted('2');

        expect(component.testTodos()).toHaveLength(0);
      });
    });

    describe('Todo Editing', () => {
      it('should update todo with new values', () => {
        const updatedTodo: Todo = {
          ...testTodos[0],
          title: 'Updated Title',
          description: 'Updated Description',
          priority: 'low'
        };

        component.onTodoEdited(updatedTodo);

        const editedTodo = component.testTodos().find(todo => todo.id === '1')!;
        expect(editedTodo.title).toBe('Updated Title');
        expect(editedTodo.description).toBe('Updated Description');
        expect(editedTodo.priority).toBe('low');
        expect(editedTodo.updatedAt).not.toEqual(testTodos[0].updatedAt);
      });

      it('should not affect other todos when editing one', () => {
        const originalTodo2 = { ...component.testTodos()[1] };
        const updatedTodo: Todo = {
          ...testTodos[0],
          title: 'Updated Title'
        };

        component.onTodoEdited(updatedTodo);

        expect(component.testTodos()[1].title).toBe(originalTodo2.title);
      });

      it('should handle editing non-existent todo gracefully', () => {
        const nonExistentTodo: Todo = {
          id: 'non-existent',
          title: 'Non Existent',
          description: 'Does not exist',
          completed: false,
          priority: 'medium',
          createdAt: new Date(),
          updatedAt: new Date(),
          tags: []
        };

        const originalTodos = [...component.testTodos()];

        component.onTodoEdited(nonExistentTodo);

        expect(component.testTodos()).toEqual(originalTodos);
      });
    });
  });

  describe('Bulk Operations Scenarios', () => {
    let mixedTodos: Todo[];

    beforeEach(() => {
      mixedTodos = [
        {
          id: '1',
          title: 'Active Todo 1',
          description: 'Description 1',
          completed: false,
          priority: 'high',
          createdAt: new Date(),
          updatedAt: new Date(),
          tags: []
        },
        {
          id: '2',
          title: 'Completed Todo',
          description: 'Description 2',
          completed: true,
          priority: 'medium',
          createdAt: new Date(),
          updatedAt: new Date(),
          tags: []
        },
        {
          id: '3',
          title: 'Active Todo 2',
          description: 'Description 3',
          completed: false,
          priority: 'low',
          createdAt: new Date(),
          updatedAt: new Date(),
          tags: []
        }
      ];
      component.testTodos.set(mixedTodos);
    });

    describe('Toggle All Operation', () => {
      it('should mark all todos as completed when not all are completed', () => {
        component.onToggleAll();

        expect(component.testTodos().every(todo => todo.completed)).toBe(true);
        component.testTodos().forEach(todo => {
          expect(todo.updatedAt).not.toEqual(mixedTodos.find(original => original.id === todo.id)?.updatedAt);
        });
      });

      it('should mark all todos as incomplete when all are completed', () => {
        // First mark all as completed
        component.testTodos.update(todos => todos.map(todo => ({ ...todo, completed: true })));

        component.onToggleAll();

        expect(component.testTodos().every(todo => !todo.completed)).toBe(true);
      });

      it('should handle empty todo list gracefully', () => {
        component.testTodos.set([]);

        expect(() => component.onToggleAll()).not.toThrow();
        expect(component.testTodos()).toHaveLength(0);
      });
    });

    describe('Clear Completed Operation', () => {
      it('should remove only completed todos', () => {
        component.onClearCompleted();

        expect(component.testTodos()).toHaveLength(2);
        expect(component.testTodos().every(todo => !todo.completed)).toBe(true);
        expect(component.testTodos().find(todo => todo.id === '2')).toBeUndefined();
      });

      it('should handle list with no completed todos', () => {
        component.testTodos.update(todos => todos.map(todo => ({ ...todo, completed: false })));

        component.onClearCompleted();

        expect(component.testTodos()).toHaveLength(3);
      });

      it('should handle list with all completed todos', () => {
        component.testTodos.update(todos => todos.map(todo => ({ ...todo, completed: true })));

        component.onClearCompleted();

        expect(component.testTodos()).toHaveLength(0);
      });
    });
  });

          id: '1',
          title: 'Active High Priority',
          description: 'Description 1',

          priority: 'high',
          createdAt: new Date(),
          updatedAt: new Date(),
          tags: []
        },
        {
          id: '2',
          title: 'Completed Medium Priority',
          description: 'Description 2',
          completed: true,
          priority: 'medium',

        },
        {

  describe('Error Handling Scenarios', () => {
    describe('Form Validation Error Recovery', () => {
      it('should recover from validation errors when valid input is provided', () => {
        // Trigger title error
        const invalidTitleEvent = createInputEvent('Invalid@#$');
        component.onTitleInput(invalidTitleEvent);
        expect(component.testFormErrors().title).not.toBe('');

        // Trigger description error
        const invalidDescEvent = createInputEvent('Invalid@#$');
        component.onDescriptionInput(invalidDescEvent);
        expect(component.testFormErrors().description).not.toBe('');

        // Fix both errors
        const validTitleEvent = createInputEvent('Valid Title');
        const validDescEvent = createInputEvent('Valid Description');
        component.onTitleInput(validTitleEvent);
        component.onDescriptionInput(validDescEvent);

        expect(component.testFormErrors().title).toBe('');
        expect(component.testFormErrors().description).toBe('');
        expect(component.testIsFormValid()).toBe(true);
      });

      it('should handle empty field validation followed by valid input', () => {
        // Clear title
        const emptyTitleEvent = createInputEvent('');
        component.onTitleInput(emptyTitleEvent);
        expect(component.testFormErrors().title).toBe('Title is required');

        // Add valid title
        const validTitleEvent = createInputEvent('Valid Title');
        component.onTitleInput(validTitleEvent);
        expect(component.testFormErrors().title).toBe('');
      });
    });

    describe('Edge Cases', () => {
      it('should handle special characters correctly', () => {
        const specialCharEvent = createInputEvent('Title with special chars !@#$%');
        component.onTitleInput(specialCharEvent);

        expect(component.testFormErrors().title).toBe('Title should only contain letters, numbers, and spaces');
      });

      it('should handle very long input strings', () => {
        const longString = 'A'.repeat(1000);
        const longTitleEvent = createInputEvent(longString);
        component.onTitleInput(longTitleEvent);

        // Should pass validation (only checks pattern, not length)
        expect(component.testFormErrors().title).toBe('');
